blobToText【Blob转文本】

描述

主要是读取Blob里文本内容 v3.3.2+

# 1.示例

二进制：
<template>
  <div class="col-md-9 col-sm-12">
    <input type="file" @change="onChangeFile" accept="text/*" />

    <div class="mt-4px max-h-400px overflow-y-auto">二进制：{{ text }}</div>
  </div>
</template>

<script>
import { blobToText } from 'sf-utils2'

export default {
  data() {
    return {
      text: ''
    }
  },

  methods: {
    // 打开ie
    async onChangeFile(e) {
      const file = e.target.files[0]
      this.text = await blobToText(file)
      console.log('blobToText转化输出结果 ===>', this.text)
      this.$message.warning(`打开控制台查看`)
    }
  }
}
</script>

# 3.源码

/**
 * 将blob对象序列化为base64
 * @param blob
 */
async function blobToText(blob: Blob): Promise<string> {
  if (!(blob instanceof Blob)) return Promise.reject('传入的参数blob非Blob对象')
  return new Promise((resolve, reject) => {
    const reader = new FileReader()
    reader.onload = () => resolve(reader.result as string)
    reader.onerror = e => reject(e)
    reader.readAsDataURL(blob)
  })
}

export default blobToText
