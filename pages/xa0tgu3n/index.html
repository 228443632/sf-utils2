# 14 stringifyCircularJSON(obj) 将包含循环引用的 JSON 对象序列化为 JSON 格式。

let a = { name: 'a' }
let b = { name: 'b' }
a.b = b
b.a = a
console.log(a)
// 输出 {"name":"a","b":{"name":"b"}}

const obj = { n: 42 }
obj.obj = obj
stringifyCircularJSON(obj) // '{"n": 42}'
