::: note 描述
从对象中根据路径获取到值
:::

## 1.示例

```js
import { merge } from '@bianpengfei/utils'

let obj = { aaa: { bbb: { ccc: 123 } } }
let prop = 'aaa.bbb'
console.log(getValueByPath(obj, prop)) // {ccc: 123}

let prop = 'aaa.bbb.ccc'
console.log(getValueByPath(obj, prop)) // 123

let obj2 = { aaa: { bbb: { ccc: 123 } }, arr: [{ name: '蔡徐坤' }] }

let prop = 'arr.0.name'
console.log(getValueByPath(obj, prop)) // '蔡徐坤'
```

## 2.入参说明

| 参数   | 说明     | 类型     | 是否必填 |
| ------ | -------- | -------- | -------- |
| object | 源对象   | `Object` | 是       |
| prop   | 属性路径 | `String` | 是       |

## 3.源码

```js
/**
 * 从对象中根据路径获取到值
 * @param {Object} object
 * @param {String} prop
 * @returns {null}
 */
function getValueByPath(object, prop) {
  prop = prop || ''
  const paths = prop.split('.')
  let current = object
  let result = null
  for (let i = 0, j = paths.length; i < j; i++) {
    const path = paths[i]
    if (!current) break

    if (i === j - 1) {
      result = current[path]
      break
    }
    current = current[path]
  }
  return result
}
```
