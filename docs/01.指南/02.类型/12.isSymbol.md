::: note 描述
判断数据类型是否是 symbol, 返回值 boolean
:::

## 1.示例

```js
import { isSymbol } from '@bianpengfei/utils'

isSymbol(new Set([1, 2, 3, {}])) // false

isSymbol(Promise.resolve()) // false

isSymbol(Symbol('s')) // true

isSymbol('1') // false

isSymbol(null) // false

isSymbol(undefined) // false

isSymbol('2s') // false

isSymbol(0) // false

isSymbol({}) // false

isSymbol([1, 2, 3]) // false

isSymbol({ name: '@bianpengfei/utils' }) // false

isSymbol(() => {}) // false
```

## 2.入参说明

| 参数 | 说明 | 类型 | 是否必填 |
| ---- | ---- | ---- | -------- |
| arg  |      | any  | 是       |

## 3.源码

```js
/**
 * 判断是否是 symbol
 * @param {string} arg
 * @returns {boolean}
 */
function isSymbol(arg) {
  return Object.prototype.toString.call(arg).slice(8, -1) === 'Symbol'
}
```
