## 1.示例

::: demo

```vue
<template>
  <div class="flex flex-col gap-8px">
    <div>文本：<input type="file" @change="onChangeFile" accept="text/*" /></div>
    <div>
      <div class="text-14px font-bold">读取到的文本内容：</div>
      <div v-if="text" class="max-h-400px overflow-auto">
        <code>{{ text }}</code>
      </div>
    </div>
  </div>
</template>

<script>
import { arrayBufferToText } from 'sf-utils2'

export default {
  data() {
    return {
      text: ''
    }
  },

  methods: {
    // 打开ie
    async onChangeFile(e) {
      const file = e.target.files[0]

      const fileReader = new FileReader()
      fileReader.readAsArrayBuffer(file)
      fileReader.onload = async () => {
        console.log('fileReader.result', fileReader.result)
        this.text = await arrayBufferToText(fileReader.result)
      }
    }
  }
}
</script>
```

:::

## 3.源码

```js
/**
 * 将二进制转成文本
 * @param arrayBuffer
 */
async function arrayBufferToText(arrayBuffer: ArrayBuffer): Promise<string> {
  const decoder = new TextDecoder('utf8', { fatal: true })
  return decoder.decode(arrayBuffer)
}

export default arrayBufferToText
```
