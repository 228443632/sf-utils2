::: note 描述
主要是将`ArrayBuffer`转化为`Base64` <Badge text="v3.3.2+" type="success" vertical="middle"/>
:::

## 1.示例

::: demo

```vue
<template>
  <div class="col-md-9 col-sm-12">
    <input type="file" @change="onChangeFile" accept="image/*" />
    <img :src="imgSrc" alt="" />
  </div>
</template>

<script>
import { base64ToBlob } from 'sf-utils2'

export default {
  data() {
    return {
      imgSrc: ''
    }
  },

  methods: {
    // 打开ie
    async onChangeFile(e) {
      const file = e.target.files[0]
      const fileReader = new FileReader()
      fileReader.readAsDataURL(file)
      fileReader.onload = async () => {
        const buffer = await file.arrayBuffer()
        if (this._blob) window.URL.revokeObjectURL(this._blob)
        this._blob = await base64ToBlob(fileReader.result, { mimeType: file.type })
        this.imgSrc = window.URL.createObjectURL(this._blob)
        console.log('打开控制台查看', this.imgSrc)
      }
    },

    beforeDestroy() {
      if (this._blob) window.URL.revokeObjectURL(this._blob)
    }
  }
}
</script>
```

## 2.入参说明

| 参数    | 说明   | 类型                                                                                                                      | 是否必填 | 默认值 |
| ------- | ------ | ------------------------------------------------------------------------------------------------------------------------- | -------- | ------ |
| base64  | base64 | `string`                                                                                                                  | 是       |        |
| options | 选项   | `TBaseBlobArrayBufferTransferOptions` ，[详情见下 👇](/pages/base64ToBlob/#_2-1-tbaseblobarraybuffertransferoptions-说明) | 否       |        |

| 参数     | 说明          | 类型     | 是否必填 | 默认值 |

## 3.源码

```js
import type { TBaseBlobArrayBufferTransferOptions } from '@/_types/_helperTreeBaseType.ts'
import base64ToArrayBuffer from '@/stream/base64ToArrayBuffer.ts'

/**
 * 将Base64字符串转换为Blob对象（自动提取MIME类型）
 * @param {string} base64 - 包含前缀的Base64字符串（如 "data:image/png;base64,..."）
 * @param options
 * @returns {Blob}
 */
async function base64ToBlob(base64: string, options?: TBaseBlobArrayBufferTransferOptions): Promise<Blob> {
  const arrayBuffer = await base64ToArrayBuffer(base64, options)
  // 4. 创建Blob对象
  return new Blob([arrayBuffer], { type: options?.mimeType || arrayBuffer['mimeType'] })
}

export default base64ToBlob
```
