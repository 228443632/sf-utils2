::: note 描述
主要是将`Blob`转化为`Base64` <Badge text="v3.3.2+" type="success" vertical="middle"/>
:::

## 1.示例

::: demo

```vue
<template>
  <div class="col-md-9 col-sm-12">
    <input type="file" @change="onChangeFile" accept="image/*" />
    <img :src="imgSrc" />
  </div>
</template>

<script>
import { blobToBase64 } from 'sf-utils2'

export default {
  data() {
    return {
      imgSrc: ''
    }
  },

  methods: {
    // 打开ie
    async onChangeFile(e) {
      const file = e.target.files[0]
      this._base64 = await blobToBase64(file)
      console.log('blobToBase64转化输出结果 ===>', this._base64)
      this.imgSrc = this._base64
      this.$message.warning(`打开控制台查看`)
    }
  }
}
</script>
```

## 2.入参说明

| 参数 | 说明      | 类型   | 是否必填 | 默认值 |
| ---- | --------- | ------ | -------- | ------ |
| blob | Blob 对象 | `Blob` | 是       |        |

:::

## 3.源码

```js
/**
 * 将blob对象序列化为base64
 * @param blob
 */
async function blobToBase64(blob: Blob): Promise<string> {
  if (!(blob instanceof Blob)) return Promise.reject('传入的参数blob非Blob对象')
  return new Promise((resolve, reject) => {
    const reader = new FileReader()
    reader.onload = () => resolve(reader.result as string)
    reader.onerror = e => reject(e)
    reader.readAsDataURL(blob)
  })
}

export default blobToBase64
```
