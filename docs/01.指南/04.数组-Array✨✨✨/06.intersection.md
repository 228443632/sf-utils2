数组转成 obj
:::

## 1.示例

```js
import { intersection } from '@bianpengfei/utils'

const list = [
 {empId: 1, empName: '蔡徐坤'},
 {empId: 2, empName: '小明'}，
 {empId: 3, empName: '小红'}，
]

// 根据属性名转化
intersection(list, 'empId')
// 输出
{ 1: {empId: 1, empName: '蔡徐坤'},2: {empId: 2, empName: '小明'}, 3: {empId: 3, empName: '小红'} }

// 不传属性名
const list = [1, 2, 3, '4']
intersection(list)
// 输出 { 1: 1, 2: 2, 3: 3, 4: '4'}


// 返回属性值为数组
intersection(list, 'empId', {valueType: 'array'})
// 输出
{ 1: [{empId: 1, empName: '蔡徐坤'}],2: [{empId: 2, empName: '小明'}], 3: [{empId: 3, empName: '小红'}] }


// 通过function，函数自定义
intersection(list, (obj, item, index) => {
    obj[item.empId] = item
})
// 输出
{ 1: {empId: 1, empName: '蔡徐坤'},2: {empId: 2, empName: '小明'}, 3: {empId: 3, empName: '小红'} }
```

```js
import intersection from '@/array/intersection'
import uniq from '@/array/uniq'
import isPrimitive from '@/base/isPrimitive'

/**
 * 获取多个数组的交集
 * @param {Array[]} array
 * @param {String|unknown} property
 * @return {[string, any]|*[]}
 */
function intersection(array, property) {
  const arrayFlatten = array.flat(Infinity) || []
  const arrayObj$property = intersection(arrayFlatten, property, { valueType: 'array' })
  const [array$0] = arrayFlatten
  const result = Object.entries(arrayObj$property).reduce((pre, [k, v]) => {
    if (v?.length > 1) {
      pre.push(...v)
    }
    return pre
  }, [])
  if (isPrimitive(array$0)) {
    return uniq(result)
  }
  return result
}

export default intersection
```
