```shell
#web-uat

#pc
upstream web-pc-uat {
    server 158.233.255.43:30547;
}

#mobile
upstream web-mobile-uat {
    server 158.233.255.43:30515;
}

#msite
upstream web-msite-uat {
    server 158.233.255.43:30516;
}

#backend-uat
upstream opr-gateway-uat {
    server 158.233.255.43:31040;
}

upstream platform-gateway-uat {
    server 158.233.255.43:31044;
}

upstream app-gateway-uat {
    server 158.233.255.43:31046;
}

upstream track-gateway-uat {
    server 158.233.255.43:31048;
}


server {
    listen 9090 ssl;
    server_name 39.185.228.190;
    set $is_mobile false;

    if ($http_user_agent ~* "(Android|iPhone|Windows Phone|UC|Kindle)") {
        set $is_mobile true;
    }


    access_log logs/access_9090.log main;
    error_log logs/error_9090.log;
    client_max_body_size 10G;

    ssl_certificate /opt/nginx/crt/tls.crt;
    ssl_certificate_key /opt/nginx/crt/tls.key;
    ssl_session_timeout 30m;
    charset utf-8; #chinese

    #pc adjust to msite
    location = / {
        proxy_pass http://web-pc-uat;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;

        if ($is_mobile = true) {
            rewrite ^/(.*)$ msite/ redirect;
        }
    }

    #m站
    location = /msite/ {
        proxy_pass http://web-msite-uat;
        client_max_body_size 10G;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        if ($is_mobile = false) {
            rewrite ^/msite/(.*)$ $scheme://$host redirect;
        }
    }

    #m站
    location ^~ /msite/ {
        proxy_pass http://web-msite-uat;
        client_max_body_size 10G;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        # rewrite ^/msite/(.*)$ /$1 break;
    }

    #pc
    location / {
        proxy_pass http://web-pc-uat;
        client_max_body_size 10G;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        #rewrite ^(.*)$ /system-maintaining/default.html permanent;
        # return 404;
    }

    # 默认页
    location ^~ /system-maintaining/ {
        alias /opt/nginx/html/;
    }


    #移动端
    location ^~ /mobile/ {
        proxy_pass http://web-mobile-uat;
        client_max_body_size 10G;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        #rewrite ^(.*)$ /system-maintaining/default.html permanent;
    }


    #uat-backend
    location /opr {
        proxy_pass http://opr-gateway-uat;
        client_max_body_size 10G;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 3600s;
    }
    location /platform {
        proxy_pass http://platform-gateway-uat;
        client_max_body_size 10G;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 3600s;
    }
    location /app {
        proxy_pass http://app-gateway-uat;
        client_max_body_size 10G;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 3600s;
    }

    location /track {
        proxy_pass http://track-gateway-uat;
        client_max_body_size 10G;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 300s;
    }

    location /zsap-pre {
        proxy_pass http://158.233.255.19:7777;
        client_max_body_size 10G;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_read_timeout 300s;
    }


    location /zsap-uat {
        proxy_pass http://158.233.255.7:9000;
        proxy_set_header Host $http_host;
        proxy_connect_timeout 15s;
        proxy_send_timeout 15s;
        proxy_read_timeout 15s;
        proxy_set_header X-Forwarded-Proto http;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
    location = /zsap-uat {
        root html;
    }
    location = /zsap-uat/ {
        root html;
    }

    location = /zsap-pre {
        root html;
    }
    location = /zsap-pre/ {
        root html;
    }


    location /zsap-sit {
        proxy_pass http://158.233.255.7:9000;
        proxy_set_header Host $http_host;
        proxy_connect_timeout 15s;
        proxy_send_timeout 15s;
        proxy_read_timeout 15s;
        proxy_set_header X-Forwarded-Proto http;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    }
    location = /zsap-sit {
        root html;
    }


    location = /zsap-sit/ {
        root html;
    }


    #      location = /CJXUzf5H3U.txt {
    #      root /opt/nginx/html;
    #     }

        root /opt/nginx/html;
    }


    location /zsap-dev {
```
