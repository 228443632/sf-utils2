removeClass

描述

移除 dom 某个 class，为了兼顾 ie，因为低版本 ie 不支持 el.classList。
支持批量移除 class， 空格隔开

# 1.示例

推荐打开控制查看 dom元素节点 给body添加 theme-mode-dark 类 给body移除 theme-mode-dark 类
给body批量添加 theme、theme-1 类 给body批量移除 theme、theme-1 类
<template>
  <div>
    <el-divider content-position="left">推荐打开控制查看 dom元素节点</el-divider>

    <el-row>
      <el-col :span="12">
        <el-button type="primary" @click="bodyAddClass('theme-mode-dark', $event)"
          >给body添加 theme-mode-dark 类</el-button
        >
      </el-col>
      <el-col :span="12">
        <el-button type="danger" @click="bodyRemoveClass('theme-mode-dark', $event)"
          >给body移除 theme-mode-dark 类</el-button
        >
      </el-col>

      <el-col :span="24">
        <div style="height: 10px; width: 100%"></div>
      </el-col>

      <el-col :span="12">
        <el-button type="primary" @click="bodyAddClass('theme theme-1', $event)"
          >给body批量添加 theme、theme-1 类</el-button
        >
      </el-col>
      <el-col :span="12">
        <el-button type="danger" @click="bodyRemoveClass('theme theme-1', $event)"
          >给body批量移除 theme、theme-1 类</el-button
        >
      </el-col>
    </el-row>
  </div>
</template>

<script>
import { addClass, removeClass } from '@bianpengfei/utils'
export default {
  methods: {
    bodyAddClass(cls = '', event) {
      addClass(document.querySelector('body'), cls)
      this.$message.success(`${event.target.innerText}添加成功，请打开控制台查看dom节点`)
    },

    bodyRemoveClass(cls = '', event) {
      removeClass(document.querySelector('body'), cls)
      this.$message.success(`${event.target.innerText}移除成功，请打开控制台查看dom节点`)
    }
  }
}
</script>

⚠️ 注意

当移除 class 类时，需要以空格隔开，比如添加两个类，class-1 和 class-2， 这时候传值 class 是
class-1 class-2

# 2.入参说明

参数 说明 类型 是否必填 默认值
el dom 元素 HTMLElement - -
cls class 名 String - -

# 3.源码

源码，点开查看 👈 
/**
 * 移除dom 某个类名
 * @param el
 * @param cls
 */
function removeClass(el, cls) {
  if (!el || !cls) return
  let classes = cls.split(' ')
  let curClass = ' ' + el.className + ' '

  for (let i = 0, j = classes.length; i < j; i++) {
    let clsName = classes[i]
    if (!clsName) continue

    if (el.classList) {
      el.classList.remove(clsName)
    } else {
      if (this.hasClass(el, clsName)) {
        curClass = curClass.replace(' ' + clsName + ' ', ' ')
      }
    }
  }
  if (!el.classList) {
    el.className = this.trim(curClass)
  }
}
