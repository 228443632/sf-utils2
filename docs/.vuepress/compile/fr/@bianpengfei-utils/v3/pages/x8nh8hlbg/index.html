isInteger

描述

判断数据类型是否是整数, 返回值 boolean

# 1.示例

import { isInteger } from '@bianpengfei/utils'

isInteger(null) // false
isInteger(3.3) // false
isInteger(3) // true
isInteger(0) // true

# 3.源码

import isNumber from './isNumber'

/**
 * 判断是否整数
 * @param {any} val
 */
function isInteger(val) {
  return isNumber(val) && val % 1 === 0
}
