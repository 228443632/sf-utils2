setStyle

描述

设置 dom style 样式，兼容 element.style[属性] = 属性值，为了兼顾 ie8 浏览器

# 1.示例

设置body背景颜色为红色 设置body背景颜色为蓝色 设置body背景颜色为transparent
<template>
  <div>
    <el-button type="primary" @click="onSetBody('red')">设置body背景颜色为红色</el-button>
    <el-button type="primary" @click="onSetBody('blue')">设置body背景颜色为蓝色</el-button>
    <el-button type="primary" @click="onSetBody('transparent')">设置body背景颜色为transparent</el-button>
  </div>
</template>

<script>
import { setStyle } from '@bianpengfei/utils'
export default {
  methods: {
    onSetBody(color) {
      setStyle(document.querySelector('body'), 'background-color', color)
    }
  }
}
</script>

# 2.入参说明

# 3.源码

源码，点开查看 👈 
import { _ieVersion } from './_constant'
import isObject from '../type/isObject'
import camelCase from '../base/camelCase'

/**
 * 设置元素样式
 * @param {HTMLElement} element dom元素
 * @param {string | object} styleName
 * @param {string} value
 */
function setStyle(element, styleName, value) {
  if (!element || !styleName) return
  if (isObject(styleName)) {
    for (let prop in styleName) {
      if (Object.prototype.hasOwnProperty.call(styleName, prop)) {
        setStyle(element, prop, styleName[prop])
      }
    }
  } else {
    styleName = camelCase(styleName)
    if (styleName === 'opacity' && _ieVersion < 9) {
      element.style.filter = isNaN(value) ? '' : 'alpha(opacity=' + value * 100 + ')'
    } else {
      element.style[styleName] = value
    }
  }
}
