downloadFile【http下载文件】

描述

根据 http url 下载远程文件到本地，返回类型是 promise
如果发现兼容性问题，请用 file-saver (opens new window)

# 1.示例

下载远程图片
<template>
  <div>
    <el-button @click="onClick">下载远程图片</el-button>
  </div>
</template>

<script>
import { downloadFile } from '@bianpengfei/utils'
const url = 'https://info.compassedu.hk/ad/225/1637553399694.png?imageView2/3/q/100|imageslim'
export default {
  methods: {
    async onClick() {
      await downloadFile({ url, filename: `${Date.now()}.png` })
      this.$message.success('下载成功')
    }
  }
}
</script>

参数 说明 类型 是否必填 默认值

# 3.源码

源码，点开查看 👈 
import getHttpBlob from './getHttpBlob'

/**
 * 下载文件
 * @param url
 * @param filename
 */
async function downloadFile({ url, filename = '' }) {
  const saveAs = (blob, filename) => {
    if (window?.navigator.msSaveOrOpenBlob) {
      navigator.msSaveBlob(blob, filename)
    } else {
      const link = document?.createElement('a')
      const body = document?.querySelector('body')
      link.href = window?.URL.createObjectURL(blob) // 创建对象url
      link.download = filename
      // fix Firefox
      link.style.display = 'none'
      body.appendChild(link)
      link.click()
      body.removeChild(link)
      window?.URL.revokeObjectURL(link.href) // 通过调用 URL.createObjectURL() 创建的 URL 对象
    }
  }
  const blob = await getHttpBlob(url)
  saveAs(blob, filename)
}
