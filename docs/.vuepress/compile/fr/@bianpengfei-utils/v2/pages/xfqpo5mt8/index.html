capitalize【中横线命名】

描述

首字母大写

# 1.示例

import { capitalize } from '@bianpengfei/utils'

capitalize('userName') // 结果：UserName
capitalize('helloWorld_name') // 结果：HelloWorld_name
capitalize('cellItem') // 结果：CellItem
capitalize('USERITEM') // 结果：u-s-e-r-i-t-e-m

# 2.入参说明

参数 说明 类型 是否必填 默认值
str 要调整的字符串 String - -

# 3.源码

源码，点开查看 👈 
import isString from '../type/isString.js'

/**
 * 首字母大写
 * @param {String}  str
 * @returns {string|*}
 */
function capitalize(str) {
  if (!isString(str)) return str
  return str.charAt(0).toUpperCase() + str.slice(1)
}

export default capitalize
