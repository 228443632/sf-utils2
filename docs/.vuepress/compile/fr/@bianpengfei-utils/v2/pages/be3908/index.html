isPromise

描述

判断数据类型是否是 promise, 返回值 boolean

# 1.示例

import { isPromise } from '@bianpengfei/utils'

isPromise(new Set([1, 2, 3, {}])) // false

isPromise(Promise.resolve()) // true

isPromise('1') // false

isPromise(null) // false

isPromise(undefined) // false

isPromise('2s') // false

isPromise(0) // false

isPromise({}) // false

isPromise([1, 2, 3]) // false

isPromise({ name: '@bianpengfei/utils' }) // false

isPromise(() => {}) // false

# 2.入参说明

参数 说明 类型 是否必填
arg any 是

# 3.源码

/**
 * 判断是否是promise
 * @param arg
 * @returns {boolean}
 */
function isPromise(arg) {
  return Object.prototype.toString.call(arg).slice(8, -1) === 'Promise'
}
