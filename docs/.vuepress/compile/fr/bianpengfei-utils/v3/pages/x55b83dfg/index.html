arrayToObj 【数组转成obj】

描述

数组转成 obj

# 1.示例

import { arrayToObj } from 'sf-utils2'

const list = [
 {empId: 1, empName: '蔡徐坤'},
 {empId: 2, empName: '小明'}，
 {empId: 3, empName: '小红'}，
]

// 根据属性名转化
arrayToObj(list, 'empId')
// 输出
{ 1: {empId: 1, empName: '蔡徐坤'},2: {empId: 2, empName: '小明'}, 3: {empId: 3, empName: '小红'} }

// 不传属性名
const list = [1, 2, 3, '4']
arrayToObj(list)
// 输出 { 1: 1, 2: 2, 3: 3, 4: '4'}


// 返回属性值为数组
arrayToObj(list, 'empId', {valueType: 'array'})
// 输出
{ 1: [{empId: 1, empName: '蔡徐坤'}],2: [{empId: 2, empName: '小明'}], 3: [{empId: 3, empName: '小红'}] }


// 通过function，函数自定义
arrayToObj(list, (obj, item, index) => {
    obj[item.empId] = item
})
// 输出
{ 1: {empId: 1, empName: '蔡徐坤'}, 2: {empId: 2, empName: '小明'}, 3: {empId: 3, empName: '小红'} }

# 2.入参说明

参数 说明 类型 是否必填
array 原数组 Array --------
property 属性名或回调函数 Array、Function --------
options 配置参数详情见下 👇 Object --------

# options 对象

参数 说明 类型 是否必填 默认值
valueType 每一项返回值类型是对象还是数组，前提建立当设置 proerty 是字符串有值时 Object、Array 否 ----

# 3.源码

源码，点开查看 👈 
import isArray from '@/base/isArray'
import isFunction from '@/base/isFunction'

/**
 * 将数组转成obj
 * @param {Object[]} array 数组
 * @param {String|Function} property 主键
 * @param {{valueType: 'object' | 'array'}} options
 * @returns {*}
 * @example
 *
 * const list = [
 *  {empId: 1, empName: '蔡徐坤'},
 *  {empId: 2, empName: '小明'}
 * ]
 * arrayToObj(list, 'empId')
 * =>
 * {
 *   1: {empId: 1, empName: '蔡徐坤'},
 *   2: {empId: 2, empName: '小明'}
 * }
 *
 * const list = [
 *  1, 2, 3, '4'
 * ]
 * arrayToObj(list)
 * =>
 * {
 *   1: 1
 *   2: 2,
 *   3: 3,
 *   4: '4'
 * }
 *
 */

function arrayToObj(array = [], property, options = { valueType: 'object' }) {
  if (isArray(array)) {
    // 如果property是Function
    if (isFunction(property)) {
      const object = {}
      array.forEach((v, vi) => property(object, v, vi))
      return object
    }
    const valueType = String(options?.valueType).toLowerCase() || 'object'
    // 非Function，且property存在
    if (property) {
      return array.reduce((pre, cur) => {
        const value = cur?.[property]
        if (value) {
          if (valueType === 'array') {
            if (!isArray(pre[value])) pre[value] = []
            pre[value].push(cur)
          } else {
            pre[value] = cur
          }
        }
        return pre
      }, {})
    }
    // property不存在时
    return array.reduce((pre, cur) => {
      if (cur) {
        if (valueType === 'array') {
          if (!isArray(pre[cur])) pre[cur] = []
          pre[cur].push(cur)
        } else {
          pre[cur] = cur
        }
      }
      return pre
    }, {})
  }
  return {}
}
export default arrayToObj
