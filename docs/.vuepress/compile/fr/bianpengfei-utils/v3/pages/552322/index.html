isString

描述

判断数据类型是否是字符串, 返回值 boolean

# 1.示例

import { isString } from 'sf-utils2'

isString('1') // true
isString('2s') // true
isString(0) // false
isString({}) // false
isString([1, 2, 3]) // false
isString({ name: 'sf-utils2' }) // false
isString(() => {}) // false

# 3.源码

import _typeof from '@/type/_typeof'

/**
 * 是否是字符串
 * @param {any} val
 * @returns {boolean}
 */
function isString(val) {
  return _typeof(val) === 'String'
}
