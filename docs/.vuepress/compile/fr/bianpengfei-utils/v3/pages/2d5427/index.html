isNumber【判断number类型】

描述

判断数据类型是否是Number类型, 返回值 boolean

# 1.示例

import { isNumber } from 'sf-utils2'

isNumber('1') // false
isNumber('2s') // false
isNumber(0) // true
isNumber({}) // false

# 3.源码

import _typeof from '@/type/_typeof'

/**
 * 判断是否是数字
 * @param {any} val
 * @returns {boolean}
 */
function isNumber(val) {
  return _typeof(val) === 'Number'
}
