::: note 描述
合并多个对象
:::

## 1.示例

```js
import { merge } from '@bianpengfei/utils'

let a = { id: 1, name: '香蕉', pId: null, order: 2 }
let b = { sort: true, name: '卞鹏飞' }
let c = { x: 10, y: 100 }

merge({}, a, b, c)

// output:
// {
//   "id": 1,
//   "name": "卞鹏飞",
//   "pId": null,
//   "order": 2,
//   "sort": true,
//   "x": 10,
//   "y": 100
// }
```

## 3.源码

```js
/**
 * 合并对象
 * @param {Object} target
 * @returns {{}}
 */
function merge() {
  let target = {}
  for (let i = 1, j = arguments.length; i < j; i++) {
    let source = arguments[i] || {}
    for (let prop in source) {
      if (source.hasOwnProperty(prop)) {
        let value = source[prop]
        if (value !== undefined) {
          target[prop] = value
        }
      }
    }
  }
  return target
}
```
