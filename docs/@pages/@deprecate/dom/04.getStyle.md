::: note 描述
获取 dom style 样式，兼容 element.style`[属性]`，为了兼顾 ie8 浏览器
:::

## 1.示例

::: demo

```vue
<template>
  <div>
    <el-button type="primary" @click="getBgColor('body')">查看body背景颜色</el-button>
    <el-button type="primary" @click="getCurButtonPadding($event)">查看当前按钮</el-button>
    <el-button type="primary" @click="getBgColor($event.target)">查看当前按钮背景颜色</el-button>
  </div>
</template>

<script>
import { getStyle, isString } from '@bianpengfei/utils'
export default {
  methods: {
    getBgColor(ctor) {
      if (isString(ctor)) {
        alert(getStyle(document.querySelector(ctor), 'background-color'))
        return
      }
      alert(getStyle(ctor, 'background-color'))
    },

    getCurButtonPadding($event) {
      console.log($event)
      alert(getStyle($event.target, 'margin-left'))
    }
  }
}
</script>
```

:::

## 2.入参说明

| 参数      | 说明         | 类型          | 是否必填 | 默认值 |
| --------- | ------------ | ------------- | -------- | ------ |
| element   | dom 元素     | `HTMLElement` | -        | -      |
| styleName | style 属性名 | `String`      | -        | -      |

## 3.源码

::: details 源码，点开查看 👈

```js
import { _ieVersion } from './_constant'
import camelCase from '../base/camelCase'

/**
 * 获取元素样式
 * @type {{(*=, *=): (null|*|number|undefined), (*=, *=): (null|*|null|undefined)}}
 */
const getStyle =
  _ieVersion < 9
    ? function (element, styleName) {
        if (!element || !styleName) return null
        styleName = camelCase(styleName)
        if (styleName === 'float') {
          styleName = 'styleFloat'
        }
        try {
          switch (styleName) {
            case 'opacity':
              try {
                return element.filters.item('alpha').opacity / 100
              } catch (e) {
                return 1.0
              }
            default:
              return element.style[styleName] || element.currentStyle ? element.currentStyle[styleName] : null
          }
        } catch (e) {
          return element.style[styleName]
        }
      }
    : function (element, styleName) {
        if (!element || !styleName) return null
        styleName = camelCase(styleName)
        if (styleName === 'float') {
          styleName = 'cssFloat'
        }
        try {
          let computed = document?.defaultView.getComputedStyle(element, '')
          return element.style[styleName] || computed ? computed[styleName] : null
        } catch (e) {
          return element.style[styleName]
        }
      }
```
